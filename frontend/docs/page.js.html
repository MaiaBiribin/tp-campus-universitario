"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import Link from "next/link";
import home from "./home.module.css";
import Card from "./components/ui/card";
import Button from "./components/ui/button";
/**
 * Renderiza la página de inicio de AulaSync.
 * Presenta una descripción general del sistema, sus principales funcionalidades
 * y un acceso a la pantalla de inicio de sesión.
 * @returns {JSX.Element} Página principal de la aplicación.
 */
export default function Home() {
    const features = [
        { titulo: "Gestión de eventos", texto: "Organizá y asigná eventos académicos" },
        { titulo: "Aulas disponibles", texto: "Asignación inteligente sin conflictos horarios" },
        { titulo: "Notificaciones", texto: "Enterate de cambios y avisos importantes" },
        { titulo: "Para todos", texto: "Estudiantes, docentes y administradores" },
    ];
    return (_jsx("div", { className: home.page, children: _jsxs("main", { className: home.main, children: [_jsxs("header", { className: home.topBar, children: [_jsxs("h1", { className: home.title, children: ["Bienvenido a ", _jsx("span", { children: "AulaSync" })] }), _jsx(Card, { className: home.infoBox, children: _jsx("p", { children: "Los usuarios deben ser aprobados por un administrador antes de ingresar al sistema." }) })] }), _jsxs("section", { className: home.hero, children: [_jsx("h2", { className: home.heroTitle, children: _jsx("span", { children: "AulaSync" }) }), _jsx("p", { className: home.heroText, children: "Organizando aulas, optimizando tiempos" }), _jsx("div", { className: home.divider })] }), _jsxs("footer", { className: home.footer, children: [_jsx("div", { className: home.features, children: features.map((feature) => (_jsxs(Card, { children: [_jsx("h3", { children: feature.titulo }), _jsx("p", { children: feature.texto })] }, feature.titulo))) }), _jsx("div", { className: home.loginButton, children: _jsx(Link, { href: "/login", children: _jsx(Button, { children: "Ingresar" }) }) })] })] }) }));
}
