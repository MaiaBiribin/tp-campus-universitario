"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { useEffect, useState } from "react";
import { getEventos } from "../services/eventos";
import Card from "./ui/card";
import cards from "../styles/cards.module.css";
import dashboard from "../styles/dashboard.module.css";
/**
 * Muestra información dinámica de un aula.
 * Consulta los eventos disponibles y determina el estado del aula:
 * ocupada si existe un evento en curso, libre si tiene eventos futuros
 * o sin eventos si no posee actividades programadas.
 * @component
 * @param {Props} props - Datos del aula.
 * @param {number} props.aulaId - Identificador del aula.
 * @param {string} props.nombre - Nombre visible del aula.
 * @returns {JSX.Element} Información del aula renderizada.
 * @example
 * <InfoAula aulaId={13} nombre="101" />
 */
export default function InfoAula({ aulaId, nombre }) {
    const [eventos, setEventos] = useState([]);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    useEffect(() => {
        async function cargar() {
            try {
                const data = await getEventos();
                const filtrados = data.filter(ev => {
                    var _a;
                    return Number((_a = ev.aula) === null || _a === void 0 ? void 0 : _a.id_aula) === Number(aulaId);
                });
                setEventos(filtrados);
            }
            catch (error) {
                setError("No se pudieron cargar los eventos del aula.");
            }
            finally {
                setCargando(false);
            }
        }
        cargar();
    }, [aulaId]);
    if (cargando) {
        return (_jsx(Card, { className: dashboard.card, children: _jsx("p", { children: "Cargando informaci\u00F3n del aula..." }) }));
    }
    if (error) {
        return (_jsx(Card, { className: dashboard.card, children: _jsx("p", { className: dashboard.error, children: error }) }));
    }
    const ahora = new Date();
    const eventoActual = eventos.find(ev => {
        const inicio = new Date(`${ev.fecha}T${ev.horaInicio}`);
        const fin = new Date(`${ev.fecha}T${ev.horaFin}`);

            ahora <= fin;
    });
    const proximo = eventos.sort((a, b) => new Date(`${b.fecha}T${b.horaInicio}`).getTime()
        -
            new Date(`${a.fecha}T${a.horaInicio}`).getTime())[0];
    return (_jsxs(Card, { className: dashboard.card, children: [_jsx("div", { className: cards.cardHeader, children: _jsxs("h3", { children: ["Aula ", nombre] }) }), eventoActual ? (_jsxs("div", { children: [_jsx("p", { children: "\uD83D\uDD34 Aula ocupada" }), _jsx("h4", { children: eventoActual.titulo }), _jsxs("p", { children: ["\uD83D\uDCC5 ", eventoActual.fecha] }), _jsxs("p", { children: ["\uD83D\uDD52 ", eventoActual.horaInicio, " - ", eventoActual.horaFin] })] }))
                :
                    proximo ? (_jsxs("div", { children: [_jsx("p", { children: "\uD83D\uDFE2 Aula libre actualmente" }), _jsx("h4", { children: "Pr\u00F3ximo evento" }), _jsx("p", { children: proximo.titulo }), _jsxs("p", { children: ["\uD83D\uDCC5 ", proximo.fecha] }), _jsxs("p", { children: ["\uD83D\uDD52 ", proximo.horaInicio] })] }))
                        :
                            (_jsx("p", { children: "\uD83D\uDFE2 No hay eventos programados" }))] }));
}
