"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { useEffect, useState } from "react";
import layout from "@/app/styles/layout.module.css";
import dashboard from "@/app/styles/dashboard.module.css";
import table from "@/app/styles/table.module.css";
import { getUsuariosHabilitados } from "@/app/services/usuarios";
import forms from "@/app/styles/forms.module.css";
/**
 * panel de usuarios habilitados
 * muestra al admin todos los usuarios del sistema que estan en estado habilitado
 *
 * @returns {JSX.Element} vista de los usuarios habilitados
 */
export default function UsuariosAdmin() {
    const [usuarios, setUsuarios] = useState([]);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    useEffect(() => {
        /**
         * carga a todos los usuarios habilitados del sistema
         */
        async function cargarUsuarios() {
            try {
                setError("");
                const usuariosData = await getUsuariosHabilitados();
                setUsuarios(usuariosData);
            }
            catch (error) {
                setError("No se pudieron cargar los usuarios habilitados.");
            }
            finally {
                setCargando(false);
            }
        }
        cargarUsuarios();
    }, []);
    if (cargando) {
        return (_jsx("main", { className: layout.main, children: _jsx("div", { className: layout.content, children: _jsx("h1", { children: "Cargando usuarios..." }) }) }));
    }
    return (_jsx("main", { className: layout.main, children: _jsxs("div", { className: layout.content, children: [_jsxs("header", { className: dashboard.header, children: [_jsxs("div", { children: [_jsx("h1", { children: "Usuarios habilitados" }), _jsx("p", { children: "Listado de usuarios activos del sistema." })] }), _jsx("div", { className: table.badge, children: usuarios.length })] }), error && (_jsxs("p", { role: "alert", className: forms.error, children: ["\u26A0\uFE0F ", error] })), usuarios.length === 0 ? (_jsx("div", { className: table.empty, children: _jsx("h2", { children: "No hay usuarios habilitados" }) })) : (_jsx("div", { className: table.tableContainer, children: _jsx("div", { className: table.tableWrapper, children: _jsxs("table", { className: table.table, children: [_jsx("thead", { children: _jsxs("tr", { children: [_jsx("th", { children: "ID" }), _jsx("th", { children: "Nombre" }), _jsx("th", { children: "Apellido" }), _jsx("th", { children: "Mail" }), _jsx("th", { children: "DNI" }), _jsx("th", { children: "Rol" })] }) }), _jsx("tbody", { children: usuarios.map((usuario) => (_jsxs("tr", { children: [_jsx("td", { children: usuario.id_usuario }), _jsx("td", { children: usuario.nombre }), _jsx("td", { children: usuario.apellido }), _jsx("td", { children: usuario.mail }), _jsx("td", { children: usuario.dni }), _jsx("td", { children: _jsx("span", { className: `${table.badge} ${table.success}`, children: usuario.rol.nombre }) })] }, usuario.id_usuario))) })] }) }) }))] }) }));
}
