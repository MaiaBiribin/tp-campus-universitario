"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import NavLinks from "./navLinks";
import { useEffect, useState } from "react";
import { getRoleFromToken } from "@/app/lib/auth";
import { useRouter } from "next/navigation";
import { logout } from "../../services/auth";
import styles from "@/app/styles/dashboard.module.css";
/**
 * Renderiza la barra lateral del sistema.
 * Muestra las opciones de navegación según el rol del usuario autenticado y permite cerrar la sesión.
 * @returns {JSX.Element} Barra lateral de navegación.
 */
export default function SideBar() {
    const router = useRouter();
    const [role, setRole] = useState(null);
    useEffect(() => {
        var _a;
        const token = (_a = document.cookie.split("; ").find(x => x.startsWith("token="))) === null || _a === void 0 ? void 0 : _a.split("=")[1];
        if (!token)
            return;
        const rol = getRoleFromToken(token);
        if (rol === "Admin")
            setRole("admin");
        if (rol === "Docente")
            setRole("docente");
        if (rol === "Estudiante")
            setRole("estudiante");
    }, []);
    /**
   * Cierra la sesión del usuario y redirige al inicio de sesión.
   * @returns {void}
   */
    function cerrarSesion() {
        logout();
        router.replace("/login");
    }
    return (_jsxs("aside", { className: styles.sidebar, children: [_jsxs("h1", { className: styles.logo, children: ["Aula", _jsx("span", { children: "Sync" })] }), role && _jsx(NavLinks, { role: role }), _jsx("button", { onClick: cerrarSesion, className: styles.logout, children: "Cerrar sesi\u00F3n" })] }));
}
