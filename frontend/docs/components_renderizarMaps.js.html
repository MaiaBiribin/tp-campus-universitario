/**
 * Renderiza mapas interactivos del edificio.
 * Permite cambiar entre pisos, seleccionar aulas
 * y mostrar información asociada al aula elegida.
 * @component
 * @returns {JSX.Element} Vista del mapa interactivo.
 */
export default function RenderizarMapas() {
    const [mapaSeleccionado, setMapaSeleccionado] = useState(Mapas.PB);
    const [aulaSeleccionada, setAulaSeleccionada] = useState(null);
    const edificio = ConfigurarMapas[mapaSeleccionado];
    return (_jsxs("div", { children: [_jsx("header", { className: "dashboardHeader", children: _jsx("div", { style: {
                        display: "flex",
                        gap: 8
                    }, children: Object.values(Mapas).map((tipo) => (_jsx(Button, { type: "button", variant: mapaSeleccionado === tipo
                            ? "primary"
                            : "danger", onClick: () => {
                            setMapaSeleccionado(tipo);
                            setAulaSeleccionada(null);
                        }, children: tipo }, tipo))) }) }), _jsxs("section", { children: [_jsx("img", { id: "mapaImagen", src: edificio.src, useMap: `#${edificio.nombreMapa}`, style: {
                            width: "100%",
                            height: "auto"
                        } }), _jsx("map", { name: edificio.nombreMapa, children: edificio.areas.map((area, i) => (_jsx("area", { shape: "poly", tabIndex: 0, coords: area.coords, alt: area.alt, href: "#", onClick: (e) => {
                                e.preventDefault();
                                setAulaSeleccionada(area);
                            } }, i))) }), aulaSeleccionada && (_jsx(InfoAula, { aulaId: aulaSeleccionada.aulaId, nombre: aulaSeleccionada.nombre }))] })] }));
}
