"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { getUsuariosPendientes, aprobarUsuario, rechazarUsuario } from "@/app/services/usuarios";
import { useEffect, useState } from "react";
import layout from "@/app/styles/layout.module.css";
import dashboard from "@/app/styles/dashboard.module.css";
import styles from "./page.module.css";
import Card from "@/app/components/ui/card";
import Button from "@/app/components/ui/button";
import forms from "@/app/styles/forms.module.css";
/**
 * Panel de solicitudes del admin.
 * Muestra las solicitudes de los nuevos usuarios que esperan a ser habilitados.
 * @returns {JSX.Element}
 */
export default function SolicitudesAdmin() {
    const [solicitudes, setSolicitudes] = useState([]);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    useEffect(() => {
        /**
         * Carga las solicitudes pendientes.
         */
        async function cargarSolicitudes() {
            try {
                setError("");
                const data = await getUsuariosPendientes();
                setSolicitudes(data);
            }
            catch (error) {
                setError("No se pudieron cargar las solicitudes.");
            }
            finally {
                setCargando(false);
            }
        }
        cargarSolicitudes();
    }, []);
    /**
     * Aprueba un usuario en específico y cambia el estado a habilitado.
     * @param {number} id  id del usario que se quiera aprobar.
     */
    async function aprobar(id) {
        try {
            setError("");
            await aprobarUsuario(id);
            setSolicitudes((prev) => prev.filter((u) => u.id_usuario !== id));
        }
        catch (error) {
            setError("No se pudo aprobar la solicitud.");
        }
    }
    /**
     * funcion para poder recharzar la solicitud de un usario
     * @param {number} id el id del usuario cuya solicitud se rechazo
     */
    async function rechazar(id) {
        try {
            setError("");
            await rechazarUsuario(id);
            setSolicitudes((prev) => prev.filter((u) => u.id_usuario !== id));
        }
        catch (error) {
            setError("No se pudo rechazar la solicitud.");
        }
    }
    if (cargando) {
        return (_jsx("main", { className: layout.main, children: _jsx("div", { className: layout.content, children: _jsx("h1", { children: "Cargando solicitudes..." }) }) }));
    }
    return (_jsx("main", { className: layout.main, children: _jsxs("div", { className: layout.content, children: [_jsxs("header", { className: dashboard.header, children: [_jsxs("div", { children: [_jsx("h1", { children: "Solicitudes de registro" }), _jsx("p", { children: "Revis\u00E1 solicitudes pendientes y aprob\u00E1 \u00FAnicamente usuarios autorizados." })] }), _jsx("div", { className: styles.counter, children: solicitudes.length })] }), error && (_jsxs("p", { role: "alert", className: forms.error, children: ["\u26A0\uFE0F ", error] })), _jsx("section", { children: solicitudes.length === 0 ? (_jsx(Card, { children: _jsxs("div", { className: styles.empty, children: [_jsx("h2", { children: "No hay solicitudes pendientes" }), _jsx("p", { children: "Cuando un usuario se registre aparecer\u00E1 ac\u00E1." })] }) })) : (_jsx("div", { className: styles.grid, children: solicitudes.map((usuario) => (_jsxs(Card, { children: [_jsxs("div", { className: styles.user, children: [_jsx("div", { className: styles.avatar, children: usuario.nombre[0] }), _jsxs("div", { children: [_jsxs("h2", { children: [usuario.nombre, " ", usuario.apellido] }), _jsx("p", { children: usuario.mail })] })] }), _jsxs("div", { className: styles.data, children: [_jsxs("div", { children: [_jsx("span", { children: "DNI" }), _jsx("strong", { children: usuario.dni })] }), _jsxs("div", { children: [_jsx("span", { children: "Estado" }), _jsx("strong", { className: styles.pending, children: "Pendiente" })] })] }), _jsxs("div", { className: styles.actions, children: [_jsx(Button, { onClick: () => aprobar(usuario.id_usuario), children: "Aprobar" }), _jsx(Button, { variant: "danger", onClick: () => rechazar(usuario.id_usuario), children: "Rechazar" })] })] }, usuario.id_usuario))) })) })] }) }));
}
