"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { useEffect, useState } from "react";
import { getAvisosPorEvento } from "@/app/services/avisos";
import { getEventos } from "@/app/services/eventos";
import dashboard from "@/app/styles/dashboard.module.css";
import forms from "@/app/styles/forms.module.css";
/**
 * Muestra los avisos más recientes del sistema.
 * Obtiene los eventos disponibles, recupera sus avisos asociados,los ordena por fecha de creación y muestra los últimos tres.
 * Gestiona estados de carga, error y ausencia de avisos mediante
 * mensajes visibles para el usuario.
 * @component
 * @returns {JSX.Element} lista de avisos recientes o mensaje de estado.
 * @example
 * <AvisosRecientes />
 */
export default function AvisosRecientes() {
    const [avisos, setAvisos] = useState([]);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    useEffect(() => {
        async function cargarAvisos() {
            try {
                setCargando(true);
                setError("");
                const eventos = await getEventos();
                const todosAvisos = [];
                for (const evento of eventos) {
                    const avisosEvento = await getAvisosPorEvento(evento.id_evento);
                    todosAvisos.push(...avisosEvento);
                }
                const ordenados = todosAvisos.sort((a, b) => new Date(b.fecha_creacion).getTime()
                    -
                        new Date(a.fecha_creacion).getTime());
                setAvisos(ordenados.slice(0, 3));
            }
            catch (error) {
                setError("No se pudieron cargar los avisos recientes.");
            }
            finally {
                setCargando(false);
            }
        }
        cargarAvisos();
    }, []);
    if (cargando) {
        return (_jsx("div", { className: dashboard.mainEvent, children: _jsx("p", { className: forms.helper, children: "Cargando avisos..." }) }));
    }
    if (error) {
        return (_jsx("div", { className: dashboard.mainEvent, children: _jsx("p", { className: forms.error, children: error }) }));
    }
    return (_jsx("div", { className: dashboard.eventGrid, children: avisos.length > 0 ? (avisos.map(aviso => {
            var _a, _b;
            return (_jsx("div", { className: dashboard.mainEvent, children: _jsxs("div", { children: [_jsx("p", { className: dashboard.label, children: "Aviso" }), _jsx("h3", { children: (_b = (_a = aviso.evento) === null || _a === void 0 ? void 0 : _a.titulo) !== null && _b !== void 0 ? _b : "Nuevo aviso" }), _jsxs("div", { className: dashboard.info, children: [_jsx("p", { children: aviso.mensaje }), _jsxs("p", { children: ["\uD83D\uDCC5 ", new Date(aviso.fecha_creacion).toLocaleDateString()] })] })] }) }, aviso.id_aviso));
        })) : (_jsx("div", { className: dashboard.mainEvent, children: _jsx("p", { children: "No hay avisos recientes." }) })) }));
}
