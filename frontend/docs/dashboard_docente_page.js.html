"use client";
import { jsx as _jsx, jsxs as _jsxs, Fragment as _Fragment } from "react/jsx-runtime";
import dashboard from "@/app/styles/dashboard.module.css";
import cards from "@/app/styles/cards.module.css";
import { useEffect, useState } from "react";
import { getEventos } from "@/app/services/eventos";
import ProximoEvento from "@/app/components/proximoEvento";
import { getAvisosPorEvento } from "@/app/services/avisos";
import forms from "@/app/styles/forms.module.css";
/**
 * Dashboard principal del docente.
 * Muestra un resumen de actividad académica: clases del día, cantidad de avisos activos y el próximo evento programado.
 * @returns {JSX.Element} Vista del panel docente.
 */
export default function DashboardDocente() {
    const [clasesHoy, setClasesHoy] = useState(0);
    const [avisosActivos, setAvisosActivos] = useState(0);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    useEffect(() => {
        /**
        * Carga la información necesaria para el panel docente.
        * Obtiene eventos, calcula las clases del día y cuenta avisos asociados.
        * @async
        * @returns {Promise<void>}
        */
        async function cargar() {
            try {
                setError("");
                setCargando(true);
                const eventos = await getEventos();
                const hoy = new Date().toISOString().split("T")[0];
                const clasesDeHoy = eventos.filter((evento) => evento.fecha === hoy);
                setClasesHoy(clasesDeHoy.length);
                let cantidadAvisos = 0;
                for (const evento of eventos) {
                    const avisos = await getAvisosPorEvento(evento.id_evento);
                    cantidadAvisos += avisos.length;
                }
                setAvisosActivos(cantidadAvisos);
            }
            catch (_a) {
                setError("No se pudo cargar la información del panel.");
            }
            finally {
                setCargando(false);
            }
        }
        cargar();
    }, []);
    if (cargando) {
        return (_jsx("p", { className: forms.helper, children: "Cargando panel..." }));
    }
    return (_jsxs(_Fragment, { children: [_jsxs("header", { className: dashboard.header, children: [_jsx("h1", { children: "Panel docente" }), _jsx("p", { children: "Consult\u00E1 tus clases, revis\u00E1 aulas asignadas y manten\u00E9 informados a tus estudiantes." })] }), error && (_jsxs("p", { role: "alert", className: forms.error, children: ["\u26A0\uFE0F ", error] })), _jsxs("section", { children: [_jsx("h2", { className: dashboard.sectionTitle, children: "Resumen del d\u00EDa" }), _jsxs("div", { className: dashboard.summaryGrid, children: [_jsxs("div", { className: cards.metric, children: [_jsx("p", { children: "Clases hoy" }), _jsx("h3", { children: clasesHoy })] }), _jsxs("div", { className: cards.metric, children: [_jsx("p", { children: "Avisos activos" }), _jsx("h3", { children: avisosActivos })] })] })] }), _jsxs("section", { children: [_jsx("h2", { className: dashboard.sectionTitle, children: "Pr\u00F3ximo evento" }), _jsx("div", { className: dashboard.eventGrid, children: _jsx(ProximoEvento, { rutaBase: "/dashboard/eventos", label: "Clase asignada" }) })] })] }));
}
