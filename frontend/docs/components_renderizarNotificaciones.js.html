"use client";
import { jsxs as _jsxs, jsx as _jsx, Fragment as _Fragment } from "react/jsx-runtime";
import { useState, useEffect } from "react";
import { TraerTodasNotificaciones, NotificacionLeida, NotificacionLeidas, } from "../services/notificaciones";
import cards from "@/app/styles/cards.module.css";
import Card from "@/app/components/ui/card";
import Button from "@/app/components/ui/button";
/**
 * Renderiza la lista de notificaciones del usuario.
 * Obtiene notificaciones periódicamente (cada 5s), permite marcar una notificación como leída o todas a la vez y muestra estados de carga y error en pantalla.
 * @component
 * @returns {JSX.Element} Lista de notificaciones con acciones asociadas.
 * @example
 * <RenderizarNotificaciones />
 */
export default function RenderizarNotifiaciones() {
    const [notificiaciones, setNotificaciones] = useState([]);
    const [error, setError] = useState(null);
    useEffect(() => {
        async function TraerNotificaciones() {
            try {
                setError(null);
                const data = await TraerTodasNotificaciones();
                setNotificaciones(data);
            }
            catch (error) {
                setError("No se pudieron cargar las notificaciones.");
            }
        }
        TraerNotificaciones();
        const intervalo = setInterval(() => {
            TraerNotificaciones();
        }, 5000);
        return () => clearInterval(intervalo);
    }, []);
    const handleMarcarNotificacion = async (Id) => {
        try {
            const notificacionLeida = await NotificacionLeida(Id);
            setNotificaciones((prev) => prev.map((N) => (N.id_notificacion === Id ? notificacionLeida : N)));
        }
        catch (error) {
            setError("No se pudo marcar la notificación como leída.");
        }
    };
    const handleMarcarNotificaciones = async () => {
        try {
            const notificacionLeidas = await NotificacionLeidas();
            setNotificaciones(notificacionLeidas);
        }
        catch (error) {
            setError("No se pudo marcar la notificación como leída.");
        }
    };
    const tienePendientes = notificiaciones.some((notif) => !notif.leida);
    if (error) {
        return (_jsxs("div", { children: [_jsxs("p", { children: ["\u26A0\uFE0F ", error] }), _jsx(Button, { onClick: () => window.location.reload(), children: "Reintentar" })] }));
    }
    return (_jsxs("div", { children: [tienePendientes && (_jsx("div", { children: _jsx(Button, { onClick: () => handleMarcarNotificaciones(), children: "Marcar todas como le\u00EDdas" }) })), _jsx("div", { className: cards.grid, children: notificiaciones.map(notif => {
                    var _a, _b;
                    return (_jsx(Card, { children: _jsxs("div", { children: [_jsx("h3", { children: (_b = (_a = notif.evento) === null || _a === void 0 ? void 0 : _a.titulo) !== null && _b !== void 0 ? _b : "Aviso" }), _jsx("p", { children: notif.mensaje }), notif.evento && (_jsxs(_Fragment, { children: [_jsxs("p", { children: ["\uD83D\uDCC5 ", notif.evento.fecha] }), _jsxs("p", { children: ["\u23F0 ", notif.evento.horaInicio] })] })), _jsx("p", { children: new Date(notif.fecha_creacion).toLocaleDateString() }), !notif.leida && (_jsx(Button, { type: "button", onClick: () => handleMarcarNotificacion(notif.id_notificacion), children: "Marcar le\u00EDda" }))] }) }, notif.id_notificacion));
                }) })] }));
}
