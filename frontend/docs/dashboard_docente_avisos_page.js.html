"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { useEffect, useState } from "react";
import layout from "@/app/styles/layout.module.css";
import dashboard from "@/app/styles/dashboard.module.css";
import buttons from "@/app/styles/buttons.module.css";
import table from "@/app/styles/table.module.css";
import forms from "@/app/styles/forms.module.css";
import { getAvisos, eliminarAviso } from "@/app/services/avisos";
/**
 * Gestiona la visualización y administración de avisos del docente.
 * Permite cargar avisos existentes, crear nuevos, editar y eliminar avisos.
 * @returns {JSX.Element} Vista de gestión de avisos.
 */
export default function Avisos() {
    const [avisos, setAvisos] = useState([]);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    const [exito, setExito] = useState("");
    const [eliminando, setEliminando] = useState(null);
    useEffect(() => {
        /**
        * Obtiene los avisos registrados y actualiza el estado de la vista.
        * @async
        * @returns {Promise<void>}
        */
        async function cargar() {
            try {
                const data = await getAvisos();
                setAvisos(data);
            }
            catch (_a) {
                setError("No se pudieron cargar los avisos.");
            }
            finally {
                setCargando(false);
            }
        }
        cargar();
    }, []);
    /**
    * Redirige a la pantalla de creación de avisos.
    * @returns {void}
    */
    function crearAviso() {
        window.location.href = "/dashboard/docente/avisos/nuevo";
    }
    /**
    * Redirige a la pantalla de edición de un aviso específico.
    * @param {number} id - Identificador del aviso a editar.
    * @returns {void}
    */
    function editar(id) {
        window.location.href = `/dashboard/docente/avisos/editar/${id}`;
    }
    /**
    * Elimina un aviso y actualiza la lista visible.
    * @async
    * @param {number} id - Identificador del aviso a eliminar.
    * @returns {Promise<void>}
    */
    async function borrar(id) {
        setError("");
        setExito("");
        setEliminando(id);
        try {
            await eliminarAviso(id);
            setAvisos(prev => prev.filter(aviso => aviso.id_aviso !== id));
            setExito("Aviso eliminado correctamente.");
        }
        catch (_a) {
            setError("No se pudo eliminar el aviso.");
        }
        finally {
            setEliminando(null);
        }
    }
    return (_jsx("main", { className: layout.main, children: _jsxs("div", { className: layout.content, children: [_jsxs("header", { className: dashboard.header, children: [_jsxs("div", { children: [_jsx("h1", { children: "Mis avisos" }), _jsx("p", { children: "Gestion\u00E1 comunicaciones para tus estudiantes." })] }), _jsx("button", { className: buttons.primary, onClick: crearAviso, children: "Crear aviso" })] }), error && (_jsx("p", { className: forms.error, children: error })), exito && (_jsx("p", { className: forms.helper, children: exito })), cargando ?
                    _jsx("p", { className: forms.helper, children: "Cargando avisos..." })
                    :
                        avisos.length === 0 ?
                            _jsxs("div", { className: table.empty, children: [_jsx("h2", { children: "No hay avisos" }), _jsx("p", { children: "Cre\u00E1 un aviso relacionado a tus eventos." })] })
                            :
                                _jsx("div", { className: table.tableContainer, children: _jsxs("table", { className: table.table, children: [_jsx("thead", { children: _jsxs("tr", { children: [_jsx("th", { children: "Mensaje" }), _jsx("th", { children: "Evento" }), _jsx("th", { children: "Fecha" }), _jsx("th", { children: "Acciones" })] }) }), _jsx("tbody", { children: avisos.map(aviso => {
                                                    var _a;
                                                    return (_jsxs("tr", { children: [_jsx("td", { children: aviso.mensaje }), _jsx("td", { children: (_a = aviso.evento) === null || _a === void 0 ? void 0 : _a.titulo }), _jsx("td", { children: new Date(aviso.fecha_creacion)
                                                                    .toLocaleDateString() }), _jsxs("td", { children: [_jsx("button", { className: buttons.secondary, onClick: () => editar(aviso.id_aviso), children: "Editar" }), _jsx("button", { className: buttons.danger, onClick: () => borrar(aviso.id_aviso), children: "Eliminar" })] })] }, aviso.id_aviso));
                                                }) })] }) })] }) }));
}
