"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { useRouter } from "next/navigation";
import { registrarUsuario } from "../services/auth";
import Link from "next/link";
import Form from "../components/ui/form";
import Button from "../components/ui/button";
import layout from "../styles/layout.module.css";
import forms from "../styles/forms.module.css";
import { useState } from "react";
/**
 * Renderiza el formulario de solicitud de registro de usuario.
 * @returns {JSX.Element} Página de registro.
 */
export default function Registrarse() {
    const ruta = useRouter();
    const [error, setError] = useState("");
    const [exito, setExito] = useState("");
    const [cargando, setCargando] = useState(false);
    /**
     * Envía la solicitud de registro al servidor.
     * @async
     * @param {React.FormEvent<HTMLFormElement>} event Evento de envío del formulario.
     * @returns {Promise<void>} Promesa que finaliza al completar el proceso.
     */
    async function MandarDatos(event) {
        event.preventDefault();
        setError("");
        setExito("");
        setCargando(true);
        const formData = new FormData(event.currentTarget);
        const datosUsuario = Object.fromEntries(formData.entries());
        try {
            const response = await registrarUsuario({
                nombre: String(datosUsuario.nombre),
                apellido: String(datosUsuario.apellido),
                mail: String(datosUsuario.mail),
                dni: String(datosUsuario.dni), contrasena: String(datosUsuario.contrasena),
            });
            if (response.ok) {
                setExito("Solicitud creada. Esperá a que un administrador apruebe tu registro.");
                setTimeout(() => {
                    ruta.push("/login");
                }, 7000);
            }
            else {
                const errorData = await response
                    .json()
                    .catch(() => ({}));
                setError(errorData.message || "Hubo un error en el registro.");
            }
        }
        catch (_a) {
            setError("Hubo un problema al conectar con el servidor.");
        }
        finally {
            setCargando(false);
        }
    }
    return (_jsx("div", { className: layout.centeredPage, children: _jsxs(Form, { title: "Crear cuenta", description: "Complet\u00E1 tus datos para solicitar acceso al sistema.", onSubmit: MandarDatos, children: [_jsxs("div", { children: [_jsx("label", { children: "Nombre" }), _jsx("input", { type: "text", name: "nombre", required: true, placeholder: "Ej: Juana", className: forms.input })] }), _jsxs("div", { children: [_jsx("label", { children: "Apellido" }), _jsx("input", { type: "text", name: "apellido", required: true, placeholder: "Ej: P\u00E9rez", className: forms.input })] }), _jsxs("div", { children: [_jsx("label", { children: "DNI" }), _jsx("input", { type: "number", name: "dni", required: true, placeholder: "Ej: 45649587", className: forms.input })] }), _jsxs("div", { children: [_jsx("label", { children: "Correo electr\u00F3nico" }), _jsx("input", { type: "email", name: "mail", required: true, placeholder: "Ej: alumno@universidad.edu", className: forms.input })] }), _jsxs("div", { children: [_jsx("label", { children: "Contrase\u00F1a" }), _jsx("input", { type: "password", name: "contrasena", required: true, placeholder: "********", className: forms.input })] }), error && (_jsx("p", { className: forms.error, children: error })), exito && (_jsx("p", { className: forms.helper, children: exito })), _jsx(Button, { type: "submit", children: "Solicitar registro" }), _jsxs("p", { className: forms.formFooter, children: ["\u00BFYa ten\u00E9s una cuenta?", _jsx(Link, { href: "/login", children: "Inici\u00E1 sesi\u00F3n" })] })] }) }));
}
