"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import layout from "@/app/styles/layout.module.css";
import forms from "@/app/styles/forms.module.css";
import Card from "@/app/components/ui/card";
import Button from "@/app/components/ui/button";
import dashboard from "@/app/styles/dashboard.module.css";
import { useCrearEvento } from "@/app/hooks/useCrearEvento";
/**
 * Creación de eventos
 * Muestra todos los datos necesarios para crear un nuevo evento.
 * @returns {JSX.Element} vista de creacion de eventos.
 */
export default function CrearEvento() {
    const evento = useCrearEvento();
    async function handleSubmit(e) {
        e.preventDefault();
        const ok = await evento.submit();
        if (ok) {
            setTimeout(() => {
                window.location.href = "/dashboard/admin/eventos";
            }, 1500);
        }
    }
    return (_jsx("main", { className: layout.main, children: _jsxs("div", { className: layout.content, children: [_jsxs("header", { className: dashboard.header, children: [_jsx("h1", { children: "Crear evento" }), _jsx("p", { children: "Registr\u00E1 una clase, parcial o final." })] }), _jsx(Card, { children: _jsxs("form", { onSubmit: handleSubmit, className: forms.form, children: [evento.error && (_jsx("p", { className: forms.error, children: evento.error })), _jsxs("div", { className: forms.row, children: [_jsxs("div", { className: forms.field, children: [_jsx("label", { className: forms.label, children: "Carrera" }), _jsxs("select", { className: forms.select, value: evento.idCarrera, onChange: (e) => evento.cambiarCarrera(e.target.value), children: [_jsx("option", { value: "", children: "Seleccionar carrera" }), evento.carreras.map((carrera) => (_jsx("option", { value: carrera.id_carrera, children: carrera.nombre }, carrera.id_carrera)))] })] }), _jsxs("div", { className: forms.field, children: [_jsx("label", { className: forms.label, children: "Materia" }), _jsxs("select", { className: forms.select, value: evento.idMateria, onChange: (e) => evento.setIdMateria(e.target.value), children: [_jsx("option", { value: "", children: "Seleccionar materia" }), evento.materias.map((materia) => (_jsx("option", { value: materia.id_materia, children: materia.nombre }, materia.id_materia)))] })] })] }), _jsxs("div", { className: forms.row, children: [_jsxs("div", { className: forms.field, children: [_jsx("label", { className: forms.label, children: "Tipo de evento" }), _jsxs("select", { className: forms.select, value: evento.idTipoEvento, onChange: (e) => evento.setIdTipoEvento(e.target.value), children: [_jsx("option", { value: "", children: "Seleccionar tipo" }), _jsx("option", { value: "1", children: "Clase" }), _jsx("option", { value: "2", children: "Parcial" }), _jsx("option", { value: "4", children: "Final" })] })] }), _jsxs("div", { className: forms.field, children: [_jsx("label", { className: forms.label, children: "Aula" }), _jsxs("select", { className: forms.select, value: evento.idAula, onChange: (e) => evento.setIdAula(e.target.value), children: [_jsx("option", { value: "", children: "Seleccionar aula" }), evento.aulas.map((aula) => (_jsx("option", { value: aula.id_aula, children: aula.nombre }, aula.id_aula)))] })] })] }), _jsx("div", { className: forms.row, children: _jsxs("div", { className: forms.field, children: [_jsx("label", { className: forms.label, children: "Fecha" }), _jsx("input", { type: "date", className: forms.input, value: evento.fecha, onChange: (e) => evento.setFecha(e.target.value) })] }) }), _jsxs("div", { className: forms.row, children: [_jsxs("div", { className: forms.field, children: [_jsx("label", { className: forms.label, children: "Hora inicio" }), _jsx("input", { type: "time", className: forms.input, value: evento.horaInicio, onChange: (e) => evento.setHoraInicio(e.target.value) })] }), _jsxs("div", { className: forms.field, children: [_jsx("label", { className: forms.label, children: "Hora fin" }), _jsx("input", { type: "time", className: forms.input, value: evento.horaFin, onChange: (e) => evento.setHoraFin(e.target.value) })] })] }), _jsx("div", { className: forms.actions, children: _jsx(Button, { type: "submit", children: "Crear evento" }) })] }) })] }) }));
}
