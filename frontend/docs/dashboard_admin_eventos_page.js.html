"use client";
import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { useEffect, useState } from "react";
import layout from "@/app/styles/layout.module.css";
import dashboard from "@/app/styles/dashboard.module.css";
import buttons from "@/app/styles/buttons.module.css";
import table from "@/app/styles/table.module.css";
import forms from "@/app/styles/forms.module.css";
import { getEventos, eliminarEvento as borrarEvento } from "@/app/services/eventos";
/**
 * panel de eventos del admin
 * muestra todos los eventos que creo el admin y tambien el link para ir a la pestana de crear un nuevo evento.
 * @returns {JSX.Element} vista de los eventos del admin
 */
export default function EventosAdmin() {
    const [eventos, setEventos] = useState([]);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    const [exito, setExito] = useState("");
    const [eliminando, setEliminando] = useState(null);
    useEffect(() => {
        /**
         * funcion para cargar los eventos a la pestaña
         *  @throws en el caso que no se pudo cargar los eventos
         */
        async function cargarEventos() {
            try {
                const data = await getEventos();
                setEventos(Array.isArray(data) ? data : []);
            }
            catch (_a) {
                setError("No se pudieron cargar los eventos.");
            }
            finally {
                setCargando(false);
            }
        }
        cargarEventos();
    }, []);
    /**
     * esta funcion redirige a la pestaña de crear un nuevo evento.
     */
    function crearEvento() {
        window.location.href = "/dashboard/admin/eventos/nuevo";
    }
    /**
     * esta funcion sirve para poder eliminar del sistema un evento
     * @param {number} id id del evento que se quiera borrar
     *  @throws en caso que no se pudo borrar el evento.
     */
    async function eliminarEvento(id) {
        setError("");
        setExito("");
        setEliminando(id);
        try {
            await borrarEvento(id);
            setEventos((prev) => prev.filter((e) => e.id_evento !== id));
            setExito("Evento eliminado correctamente.");
        }
        catch (_a) {
            setError("No se pudo eliminar el evento. Puede tener información asociada.");
        }
        finally {
            setEliminando(null);
        }
    }
    return (_jsx("main", { className: layout.main, children: _jsxs("div", { className: layout.content, children: [_jsxs("header", { className: dashboard.header, children: [_jsxs("div", { children: [_jsx("h1", { children: "Eventos acad\u00E9micos" }), _jsx("p", { children: "Gestion\u00E1 clases, parciales y finales." })] }), _jsx("button", { className: buttons.primary, onClick: crearEvento, children: "Crear evento" })] }), error && (_jsx("p", { className: forms.error, children: error })), exito && (_jsx("p", { className: forms.success, children: exito })), eventos.length === 0 ? (_jsxs("div", { className: table.empty, children: [_jsx("h2", { children: "No hay eventos creados" }), _jsx("p", { children: "Cre\u00E1 tu primer evento acad\u00E9mico." })] })) : (_jsx("div", { className: table.tableContainer, children: _jsx("div", { className: table.tableWrapper, children: _jsxs("table", { className: table.table, children: [_jsx("thead", { children: _jsxs("tr", { children: [_jsx("th", { children: "Evento" }), _jsx("th", { children: "Carrera" }), _jsx("th", { children: "Fecha" }), _jsx("th", { children: "Horario" }), _jsx("th", { children: "Aula" }), _jsx("th", { children: "Tipo" }), _jsx("th", { children: "Acciones" })] }) }), _jsx("tbody", { children: eventos.map((evento) => {
                                        var _a, _b, _c, _d, _e;
                                        return (_jsxs("tr", { children: [_jsx("td", { children: (_a = evento.materia) === null || _a === void 0 ? void 0 : _a.nombre }), _jsx("td", { children: (_c = (_b = evento.materia) === null || _b === void 0 ? void 0 : _b.carrera) === null || _c === void 0 ? void 0 : _c.nombre }), _jsx("td", { children: evento.fecha }), _jsxs("td", { children: [evento.horaInicio, " - ", evento.horaFin] }), _jsx("td", { children: (_d = evento.aula) === null || _d === void 0 ? void 0 : _d.nombre }), _jsx("td", { children: _jsx("span", { className: `${table.badge} ${table.info}`, children: (_e = evento.tipoEvento) === null || _e === void 0 ? void 0 : _e.nombre }) }), _jsx("td", { children: _jsx("div", { className: table.actions, children: _jsx("button", { className: buttons.danger, disabled: eliminando === evento.id_evento, onClick: () => eliminarEvento(evento.id_evento), children: eliminando === evento.id_evento
                                                                ? "Eliminando..."
                                                                : "Eliminar" }) }) })] }, evento.id_evento));
                                    }) })] }) }) }))] }) }));
}
