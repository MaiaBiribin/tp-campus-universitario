"use client";
import { jsx as _jsx, jsxs as _jsxs, Fragment as _Fragment } from "react/jsx-runtime";
import layout from "@/app/styles/layout.module.css";
import dashboard from "@/app/styles/dashboard.module.css";
import styles from "./page.module.css";
import Card from "@/app/components/ui/card";
import Button from "@/app/components/ui/button";
import forms from "@/app/styles/forms.module.css";
import { obtenerUsuariosDisponibles, } from "@/app/services/usuarios";
import { useAsignacionAcademica, } from "@/app/hooks/useAsignacionAcademica";
/**
 * Renderiza la gestión de asignaciones académicas.
 * Permite seleccionar una carrera, visualizar sus materias e
 * inscribir usuarios en la materia elegida.
 * @returns {JSX.Element} Vista de asignación académica.
 */
export default function AcademicoAdmin() {
    const { carreras, usuariosInscriptos, materias, materiaSeleccionada, setMateriaSeleccionada, usuarios, usuariosSeleccionados, cargando, error, exito, cambiarCarrera, cargarUsuariosInscriptos, toggleUsuario, inscribirUsuarios, } = useAsignacionAcademica();
    if (cargando) {
        return (_jsx("main", { className: layout.main, children: _jsx("div", { className: layout.content, children: _jsx("p", { className: forms.helper, children: "Cargando informaci\u00F3n acad\u00E9mica..." }) }) }));
    }
    return (_jsx("main", { className: layout.main, children: _jsxs("div", { className: layout.content, children: [_jsx("header", { className: dashboard.header, children: _jsxs("div", { children: [_jsx("h1", { children: "Asignaci\u00F3n acad\u00E9mica" }), _jsx("p", { children: "Visualiz\u00E1 carreras y materias. Inscrib\u00ED a los usuarios." })] }) }), error && (_jsx("p", { className: forms.error, children: error })), exito && (_jsx("p", { className: forms.helper, children: exito })), _jsxs("section", { className: styles.grid, children: [_jsxs(Card, { children: [_jsx("h2", { children: "Carreras" }), _jsx("div", { className: styles.list, children: carreras.map((carrera) => (_jsx(Button, { onClick: () => cambiarCarrera(carrera.id_carrera), children: carrera.nombre }, carrera.id_carrera))) })] }), _jsxs(Card, { children: [_jsx("h2", { children: "Materias" }), _jsx("div", { className: styles.list, children: materias.map((materia) => (_jsx("button", { className: (materiaSeleccionada === null || materiaSeleccionada === void 0 ? void 0 : materiaSeleccionada.id_materia) ===
                                            materia.id_materia
                                            ? styles.active
                                            : styles.item, onClick: () => {
                                            setMateriaSeleccionada(materia);
                                            cargarUsuariosInscriptos(materia.id_materia);
                                        }, children: materia.nombre }, materia.id_materia))) })] }), _jsxs(Card, { children: [_jsx("h2", { children: materiaSeleccionada
                                        ? `Inscribir en ${materiaSeleccionada.nombre}`
                                        : "Seleccioná una materia" }), materiaSeleccionada && (_jsxs(_Fragment, { children: [_jsx("div", { className: styles.usersList, children: obtenerUsuariosDisponibles(usuarios, usuariosInscriptos)
                                                .map((usuario) => (_jsxs("label", { className: styles.userRow, children: [_jsx("input", { type: "checkbox", checked: usuariosSeleccionados.includes(usuario.id_usuario), onChange: () => toggleUsuario(usuario.id_usuario) }), usuario.nombre, " ", usuario.apellido, " · DNI ", usuario.dni] }, usuario.id_usuario))) }), usuariosInscriptos.length > 0 && (_jsxs("div", { className: styles.usersList, children: [_jsx("h3", { children: "Usuarios inscriptos" }), usuarios.filter(usuario => usuariosInscriptos.includes(usuario.id_usuario)).map(usuario => (_jsxs("p", { className: styles.userRow, children: [usuario.nombre, " ", usuario.apellido, " · DNI ", usuario.dni] }, usuario.id_usuario)))] })), _jsx("div", { className: styles.actions, children: _jsx(Button, { onClick: () => inscribirUsuarios(materiaSeleccionada.id_materia), children: "Inscribir seleccionados" }) })] }))] })] })] }) }));
}
