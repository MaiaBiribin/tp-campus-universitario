"use client";
import { jsx as _jsx, jsxs as _jsxs, Fragment as _Fragment } from "react/jsx-runtime";
import Link from "next/link";
import dashboard from "@/app/styles/dashboard.module.css";
import cards from "@/app/styles/cards.module.css";
import { getEventos } from "@/app/services/eventos";
import { getUsuariosPendientes } from "@/app/services/usuarios";
import { useEffect, useState } from "react";
import forms from "@/app/styles/forms.module.css";
/**
 * Panel principal de administración.
 * Muestra un resumen de eventos del día, solicitudes pendientes y aulas ocupadas.
 * Consume datos desde servicios de eventos y usuarios.
 * @returns {JSX.Element} dashboard administrativo renderizado.
 */
export default function DashboardAdmin() {
    const [eventosHoy, setEventosHoy] = useState(0);
    const [solicitudes, setSolicitudes] = useState(0);
    const [aulasOcupadas, setAulasOcupadas] = useState(0);
    const [cargando, setCargando] = useState(true);
    const [error, setError] = useState("");
    useEffect(() => {
        /**
         * Carga de datos para el dashboard del admin.
         */
        async function cargarDashboard() {
            try {
                setCargando(true);
                setError("");
                const pendientes = await getUsuariosPendientes();
                setSolicitudes(pendientes.length);
                const eventos = await getEventos();
                const hoy = new Date().toISOString().split("T")[0];
                const eventosDeHoy = eventos.filter((evento) => evento.fecha === hoy);
                setEventosHoy(eventosDeHoy.length);
                const aulas = new Set(eventosDeHoy.map((evento) => { var _a; return (_a = evento.aula) === null || _a === void 0 ? void 0 : _a.id_aula; }));
                setAulasOcupadas(aulas.size);
            }
            catch (_a) {
                setError("No se pudo cargar la información del panel.");
            }
            finally {
                setCargando(false);
            }
        }
        cargarDashboard();
    }, []);
    if (cargando) {
        return (_jsx("p", { className: forms.helper, children: "Cargando panel..." }));
    }
    if (error) {
        return (_jsx("p", { className: forms.error, children: error }));
    }
    return (_jsxs(_Fragment, { children: [_jsxs("header", { className: dashboard.header, children: [_jsx("h1", { children: "Panel de administraci\u00F3n" }), _jsx("p", { children: "Gestion\u00E1 eventos, aulas, usuarios y la organizaci\u00F3n acad\u00E9mica desde un \u00FAnico lugar." })] }), _jsxs("section", { children: [_jsx("h2", { className: dashboard.sectionTitle, children: "Resumen r\u00E1pido" }), _jsxs("div", { className: dashboard.summaryGrid, children: [_jsxs("div", { className: cards.metric, children: [_jsx("p", { children: "Eventos hoy" }), _jsx("h3", { children: eventosHoy })] }), _jsxs("div", { className: cards.metric, children: [_jsx("p", { children: "Solicitudes pendientes" }), _jsx("h3", { children: solicitudes })] }), _jsxs("div", { className: cards.metric, children: [_jsx("p", { children: "Aulas ocupadas" }), _jsx("h3", { children: aulasOcupadas })] })] })] }), _jsxs("section", { children: [_jsx("h2", { className: dashboard.sectionTitle, children: "Accesos r\u00E1pidos" }), _jsx("div", { className: dashboard.quickGrid, children: [
                            {
                                titulo: "Crear evento",
                                desc: "Crear clases, parciales o finales",
                                ruta: "/dashboard/admin/eventos/nuevo",
                            },
                            {
                                titulo: "Académico",
                                desc: "Gestionar carreras y materias",
                                ruta: "/dashboard/admin/academico",
                            },
                        ].map((item) => (_jsxs(Link, { href: item.ruta, className: cards.quickCard, children: [_jsx("h3", { children: item.titulo }), _jsx("p", { children: item.desc })] }, item.ruta))) })] })] }));
}
